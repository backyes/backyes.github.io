1.2 核心设计哲学

Zero-Buffer 范式：通过 HCCL 共享内存 (shmem) 实现跨卡直接读写，完全消除中间 buffer 拷贝。每个 rank 预分配 100MB IPC buffer，前 2MB 为 flag 区，后 98MB 为数据区。
Flag-Based Sync：基于 magic + phase 的 64-bit flag 轮询替代传统 barrier。高 32 bit 为 magic（单调递增区分不同 launch），低 32 bit 为 value/phase。
Ping-Pong Buffer：magic % 2 选择 buffer，避免连续 kernel 间的数据冲突。同一 rank 上 kernel 严格串行，magic 单调递增保证 buffer 安全复用。
AIC/AIV 异构协作：Fused Deep MoE 中 Cube 引擎 (AIC) 做矩阵乘，Vector 引擎 (AIV) 做后处理 (SwiGLU/Quant/Dequant)，通过 CrossCoreFlag + Callback 精确同步。

2.2.2 zero_buffer_sync_flag.h — 核心同步机制

class ZeroBufferSyncFlag {
    constexpr uint32_t FLAG_SLOT_SIZE = 32U;        // 每个 flag slot 32B
    constexpr uint32_t FLAG_ELEM_NUM = 4U;          // 4 × int64_t
    constexpr uint64_t FLAG_AREA_BASE = 100UL * 1024UL;  // flag 区起始: 100KB
    constexpr uint32_t MAGIC_SLOT_ALIGN = 512U;     // 每 core 512B 对齐
    constexpr uint32_t MAX_CORE_NUM_SYNC = 48U;     // 最大 48 核

    // 每个 kernel 入口调用一次，获取唯一 magic
    __aicore__ inline void IncrementMagic() {
        GM_ADDR magicAddr = flagBaseGM_ + coreIdx_ * MAGIC_SLOT_ALIGN;
        DataCacheCleanAndInvalid(magicGT);          // 读前 cache flush
        magic_ = static_cast(magicGT(0));
        magicGT(0) = static_cast(magic_) + 1ULL;  // 自增
        DataCacheCleanAndInvalid(magicGT);          // 写后 cache flush
        barrierSeq_ = 0;
    }

    // All-to-all barrier: SyncAll → Write flags → Wait flags → SyncAll
    __aicore__ inline void BarrierAll() {
        SyncAll<true>();  // Step 1: 卡内同步
        // Step 2a: 向所有对端写 barrier flag
        for (uint32_t r = startRank; r < endRank; r++) {
            DataCopy(remoteSlot, ubFlag, FLAG_ELEM_NUM);
        }
        // Step 2b: 等待所有对端的 barrier flag
        for (uint32_t r = startRank; r < endRank; r++) {
            do { DataCopy(ubFlag, localSlot, FLAG_ELEM_NUM); } 
            while (ubFlag.GetValue(0) != barrierFlag);
        }
        SyncAll<true>();  // Step 3: 卡内同步
    }
};

2.5 moe_dispatch_normal_zb — Normal Dispatch 数据写入

__aicore__ inline void QuantProcess() {
    // 1. BF16/FP16 → FP32
    Cast(floatLocalTemp, xInTensor, RoundMode::CAST_NONE, h);
    // 2. 计算动态 scale: scale = 127.0 / max(|x|)
    Abs(floatLocalAbsTemp, floatLocalTemp, h);
    ReduceMaxInplace(floatLocalAbsTemp, h);
    dynamicScale = float(127.0) / (floatLocalAbsTemp.GetValue(0) + 1e-12f);
    // 3. 量化: x * scale → int32 → int16 → int8
    Muls(floatLocalTemp, floatLocalTemp, dynamicScale, h);
    Cast(int32LocalTemp, floatLocalTemp, RoundMode::CAST_RINT, h);
    Cast(halfLocalTemp, int32LocalTemp, RoundMode::CAST_ROUND, h);
    Cast(xOutTensor, halfLocalTemp, RoundMode::CAST_TRUNC, h);
    // 4. 保存 scale 到 token 末尾
    floatLocalTemp.SetValue(hUBAlignSize / sizeof(float), float(1.0) / dynamicScale);
}

3.2.1 LocalLayout — 本地 Expert Count 统计

__aicore__ inline void LocalLayout() {

3.2.2 SetSyncFlag / WaitSyncFlag — 细粒度跨卡同步

__aicore__ inline void SetSyncFlag() {
    // 1. 准备 flag 值 (1.0f 的 IEEE754 表示 = 0x3F800000)
    LocalTensor<int32_t> statusTensor = syncStatusBuf_.Get<int32_t>();
    Duplicate<int32_t>(statusTensor, 0, rankNumPerBlock_ * 8);
    uint64_t mask[2] = {0x101010101010101, 0};
    Duplicate<int32_t>(statusTensor, 0x3F800000, mask, statusCntAlign / 8, 1, 8);
    // 2. 向每个目标 rank 的元数据区写入 flag
    for (uint32_t targetRankId = curBlockStartRankId_; targetRankId < curBlockEndRankId_; ++targetRankId) {
        auto ptr = GetSyncFlagAddrByRankId(targetRankId) + epRankId_ * DISPATCH_SYNC_STATE_OFFSET;
        gmRemoteStatusGT.SetGlobalBuffer((__gm__ int32_t *)(ptr));
        DataCopy<int32_t>(gmRemoteStatusGT, statusTensor[...], 8UL);
    }
    SyncFunc<AscendC::HardEvent::MTE3_S>();
}

__aicore__ inline void WaitSyncFlag() {
    float compareTarget = 1.0f * rankNumPerBlock_;
    while ((sumOfFlag < minTarget) || (sumOfFlag > maxTarget)) {
        DataCopy(statusFp32Tensor, statusFp32TensorGT[...], intriParams);
        SyncFunc<AscendC::HardEvent::MTE2_V>();
        ReduceSum(statusSumOutTensor, statusFp32Tensor, gatherMaskOutTensor, mask, rankNumPerBlock_, 1);
        sumOfFlag = statusSumOutTensor.GetValue(0);
    }
    // 清理 flag 为下一轮做准备
    Duplicate<int32_t>(cleanStateTensor, 0, duplicateMask, Ceil(rankNumPerBlock_, 8), 1, 8);
    DataCopy(statusFp32TensorGT[...], cleanStateTensor.ReinterpretCast<float>(), intriOutParams);
}
同步协议关键: 每个 AIV 负责向 rankNumPerBlock_ 个远程 rank 写入 flag。使用 ReduceSum 聚合所有 flag 值，当和等于预期值时退出轮询。flag 值为 IEEE754 float 1.0f，利用浮点比较避免整数溢出问题。

3.3.1 CopyValidExpandXToRemote — 本地 Expert 数据拷贝

__aicore__ inline void CopyValidExpandXToRemote() {
    if (!needCopyExpandX_) return;
    SplitToCore(validTokenNum_, aivNum_, startTokenId, endTokenId, sendTokenNum);
    constexpr uint32_t COPY_BUFFER_NUM = 2U;
    tpipe_->InitBuffer(copyInBuf_, COPY_BUFFER_NUM * GM2GM_COPY_BLOCK_SIZE);
    for (uint64_t i = 0; dataSizeRemain > 0U; ++i) {
        uint32_t copySize = min(dataSizeRemain, GM2GM_COPY_BLOCK_SIZE);
        uint32_t bufferOffset = (i & 1U) * GM2GM_COPY_BLOCK_SIZE;
        event_t eventId = (i & 1U) ? EVENT_ID0 : EVENT_ID1;
        // 乒乓流水线: MTE2 读 → MTE3 写
        AscendC::WaitFlag<HardEvent::MTE3_MTE2>(eventId);

        AscendC::SetFlag<HardEvent::MTE2_MTE3>(eventId);
        AscendC::WaitFlag<HardEvent::MTE2_MTE3>(eventId);
        DataCopyPad(expandXByteGT[dstOffset], copyLocal[bufferOffset], copyParams);
        AscendC::SetFlag<HardEvent::MTE3_MTE2>(eventId);
        srcOffset += copySize; dstOffset += copySize; dataSizeRemain -= copySize;
    }
}

Dispatch 使用 DISPATCH_SYNC_FLAG_BASE_OFFSET = FLAG_AREA_BASE + 64KB

Dispatch 和 Combine 使用分离的 flag 区域，因为某些 rank 可能已经进入 Combine 而其他 rank 仍在 Dispatch，共享 flag slot 会导致竞争。

void L1TileMmad(L1TileMmadParams const &params) {
    uint32_t mPartLoop = CeilDiv<L0TileShape::M>(params.mRound);
    uint32_t nPartLoop = CeilDiv<L0TileShape::N>(params.nRound);
    uint32_t kPartLoop = CeilDiv<L0TileShape::K>(params.kActual);

    for (mPartIdx = 0; mPartIdx < mPartLoop; ++mPartIdx) {
        for (kPartIdx = 0; kPartIdx < kPartLoop; ++kPartIdx) {
            // 1. L1 → L0A (A 矩阵分块)
            copyL1ToL0A(l0ATile, l1ATile, layoutAInL0, L1A_LAYOUT);
            for (nPartIdx = 0; nPartIdx < nPartLoop; ++nPartIdx) {
                // 2. L1 → L0B (B 矩阵分块)
                copyL1ToL0B(l0BTile, l1BTile, layoutBInL0, L1B_LAYOUT);
                // 3. L0C = L0A × L0B (Cube 引擎矩阵乘)
                tileMmad(l0CTile, l0ATile, l0BTile, 
                         mPartActual, nPartActual, kPartActual, initC, unitFlag);
            }
        }
    }
    // 4. K 循环结束后：L0C → GM (写回)
    if (params.isKLoopLast) {
        params.callbackBeforeFixpipe();  // 通知 AIV 可以开始 epilogue
        copyL0CToGm(params.gmBlockC, l0CTensor, layoutCInGm, layoutCInL0, 0b11);
        params.callbackAfterFixpipe();   // 通知 AIV epilogue 完成
    }
}

FlexRetCode FlexAlloc::alloc(length_t req_size, size_t &offset, size_t &alloc_size) {
    alloc_size = round_up_bits(req_size, OBJ_SIZE_BITS);  // 64 字节对齐
    auto sizeIt = m_free_ranges_by_size.lower_bound(alloc_size);  // 找第一个 >= 请求的
    if (sizeIt == end) {
        if (get_total_free_mem() >= alloc_size)
            return FLEX_RC_ERR_FRAGMENTATION;  // 总量够但碎片化了
        else
            return FLEX_RC_ERR_NO_MEM;         // 真的没内存了
    }
    // 分裂剩余空间
    if (blkSz > alloc_size) {
        m_free_ranges_by_offset.emplace(offset + alloc_size, Range{blkSz - alloc_size});
    }
}

    while (newStart + newSize == rangeStart) { newSize += rangeSize; }

}

8. 关键设计洞察与跨章节主题

8.1 Flag-Based Sync vs. 传统 Barrier

传统方式 (HCCL 集合通信)：标准化但延迟高，需要额外通信开销。
Flag-Based Sync (CAM zero-buffer)：通过 shmem flag 轮询实现跨卡同步，延迟低 (RDMA write + poll)，无需额外通信。
关键保证: Magic 单调递增 ⇒ 任何历史 kernel 的 stale flag 都会被 (v & MAGIC_MASK) != (checkValue & MAGIC_MASK) 拒绝。

8.2 Detour 模式的核心思想

核心洞察: 当通信 group 是全集的子集时，非通信 rank 的 IPC 内存可以作为"免费"的中转缓冲区。 
传统方式（HCCL）：
  Rank 0 ──RDMA──▶ Rank 1（需全局 barrier，所有 rank 等待）

Detour 方式：
  Rank 0 ──GM write──▶ Rank 2（detour）──GM read──▶ Rank 1
  （仅需子集同步，Rank 3 不受影响）
适用条件: 通信 rank 数 < 总 rank 数，数据量 < 1MB，通信频率高需要低延迟。

8.3 MC2 子通道的利用

e2a/a2e 虽然绕过 HCCL 的标准集合通信 API，但仍然使用 MC2（Memory-Core Communication） 子通道的 tiling 配置：

AscendC::Mc2CcTilingConfig mc2CcTilingConfig(groupEp, opType1, algConfigAllToAllStr);
mc2CcTilingConfig.GetTiling(tiling->mc2InitTiling);


这表明 MC2 是 HCCL 内部的硬件通信调度层，提供 fullmesh/pairwise 等拓扑算法。CAM 算子直接配置 MC2 tiling，利用其底层通信能力，但自主管理数据搬运和同步。这是一种"半绕过"策略：复用硬件能力，但规避软件调度开销。

8.5 通信隐藏策略




Fused Deep MoE 的核心创新：将 AlltoAll 通信与 GMM 计算重叠。


传统方式: [====Dispatch====] [GMM1] [SwiGLU] [GMM2] [====Combine====]
          ↑ 通信暴露 ↑                              ↑ 通信暴露 ↑

融合方式: [Dispatch+GMM1] [SwiGLU+GMM2] [Combine]
          ↑ 通信隐藏 ↑    ↑ 计算重叠 ↑    ↑ 通信隐藏 ↑
          
          AlltoAll 发送在 GMM1 计算期间完成
          AlltoAll 接收在 GMM2 计算期间完成

8.7 量化融合



传统方式：
  GMM(INT8) → INT32 → Cast FP32 → ×scale → SwiGLU → Quant → INT8
  ↑ 多步分离 ↑

  INT32 → Cast FP32 → ×weight_scale → ×16 → +low → +assistMatrix → ×per_token → SwiGLU → Quant
