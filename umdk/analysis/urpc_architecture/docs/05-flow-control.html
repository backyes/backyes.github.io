原子 vs 非原子操作

流控窗口支持两种实现，通过 use_atomic_window 配置选择：

类型实现适用场景
非原子直接读写 uint16_t单线程收发
原子__atomic_compare_exchange_n CAS多线程并发收发
 
// 原子操作实现（umq_ub_flow_control.c）
static inline uint16_t remote_rx_window_inc_atomic(struct ub_flow_control *fc, uint16_t new_win) {
    uint16_t after, before = __atomic_load_n(&fc->remote_rx_window, __ATOMIC_RELAXED);
    do {
        win_sum = before + new_win;
        if (win_sum > UINT16_MAX) { /* error */ break; }
        after = (uint16_t)win_sum;
    } while (!__atomic_compare_exchange_n(&fc->remote_rx_window, &before, after,
                                           true, __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE));
    return after;
}
