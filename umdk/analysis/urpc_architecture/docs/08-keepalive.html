保活消息格式

保活消息复用 URPC REQ 头格式，使用专用 function ID：

// protocol.h:
#define URPC_KEEPALIVE_FUNCTION_ID  (0x002001000005)
#define URPC_KEEPALIVE_HDR_SIZE     (sizeof(urpc_req_head_t) + sizeof(urpc_keepalive_head_t))

// keepalive_head 结构：
typedef struct urpc_keepalive_head {
    uint8_t version : 4;
    uint8_t is_rsp : 1;          // 是否为响应
    uint8_t status;              // 处理结果
    uint16_t l_qid;              // 本地队列 ID
    uint32_t server_channel : 24;
    uint32_t rsvd[9];            // 最后 28B 用于加密头
} urpc_keepalive_head_t;
