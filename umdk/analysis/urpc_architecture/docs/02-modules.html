函数职责
urpc_func_call()Client 发起调用：选择 local/remote queue → 分配 req_entry → 填充 REQ 头 → 加密 → 发送
urpc_func_poll()轮询完成事件：遍历所有 queue → 收集 TX/RX CQE → 构造 poll_msg
urpc_func_return()Server 返回结果：填充 RSP 头 → 加密 → 发送 response
urpc_ref_read()引用读取：Worker 通过地址拉取远端参数
process_tx_msg()处理 TX CQE：REQ/RSP/ACK/READ 完成事件分发
process_req_cqe()REQ CQE 处理：根据 call_mode 决定释放 ctx/等待 RSP
tx_ctx_try_put()tx_ctx 双次释放：确保 TX CQE 和 ACK/RSP 都完成才释放

protocol.h — 协议定义（~600 行）

代码结构：protocol.h 定义 URPC 的所有协议消息格式和编解码函数。

// REQ 头（32B + DMA table）
struct urpc_req_head {
    uint64_t version : 4;
    uint64_t type : 4;
    uint64_t ack : 1;                // 是否需要 ACK
    uint64_t arg_dma_count : 5;      // DMA 地址表项数
    uint64_t function : 48;          // 函数 ID
    uint32_t req_total_size;         // 请求总大小
    uint32_t req_id;                 // 请求唯一 ID
    uint32_t client_channel : 24;    // 客户端通道 ID
    uint32_t function_defined : 8;   // 自定义字段
    arg_dma_t ub_dma[0];             // DMA 地址表（灵活数组）
};

// ACK 头（12B）
struct urpc_ack_head {
    uint8_t version : 4;
    uint8_t type : 4;
    uint16_t req_id_range;
    uint32_t req_id;
    uint32_t client_channel : 24;
};

// RSP 头（16B + offset array）
struct urpc_rsp_head {
    uint8_t version : 4;
    uint8_t type : 4;
    uint8_t status;                  // 执行状态
    uint16_t req_id_range;
    uint32_t req_id;
    uint32_t client_channel : 24;
    uint32_t function_defined : 8;
    uint32_t response_total_size;
    uint32_t return_data_offset[0];
};
