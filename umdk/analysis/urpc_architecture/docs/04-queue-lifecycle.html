// queue_send_recv.c: send_recv_bind_queue()
int send_recv_bind_queue(queue_t *l_queue, queue_t *r_queue) {
    // 1. 导入远端 jetty
    imported_queue->tjetty = urma_import_jetty(urma_ctx, &rjetty, &token);

    // 2. 绑定 jetty（建立连接）
    urma_bind_jetty(local_queue->jetty, imported_queue->tjetty);

    // 3. 记录远端 jetty_id
    local_queue->remote_jetty_id = imported_queue->tjetty->id;
    local_queue->is_binded = TRUE;
}

// 预填充 RX buffer（bind 后）
static int umq_ub_prefill_rx_buf(ub_queue_t *queue) {
    do {
        qbuf = umq_buf_alloc(queue->rx_buf_size, batch_count, 0, NULL);
        umq_ub_post_rx_inner_impl(queue, qbuf, &bad_buf);
        require_rx_count -= cur_batch_count;
    } while (require_rx_count > 0);
}
