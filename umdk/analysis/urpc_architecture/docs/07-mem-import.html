// umq_ub.c: umq_ub_data_plan_import_mem()
int umq_ub_data_plan_import_mem(uint64_t umqh_tp, umq_buf_t *rx_buf, uint32_t ref_seg_num) {
    // 1. 解析接收缓冲区头部的 umq_imm_head
    umq_imm_head_t *umq_imm_head = (umq_imm_head_t *)rx_buf->buf_data;

    // 2. 解析 import_mempool_info 数组
    ub_import_mempool_info_t *import_mempool_info =
        (ub_import_mempool_info_t *)(rx_buf->buf_data + sizeof(umq_imm_head_t) + ref_seg_num * sizeof(ub_ref_sge_t));

    // 3. 逐个导入内存池
    for (uint32_t i = 0; i < umq_imm_head->mempool_num; i++) {
        // 检查是否已导入
        if (queue->imported_tseg_list[import_mempool_info[i].mempool_id] != NULL) {
            umq_ub_import_mem_done(queue, import_mempool_info[i].mempool_id);
            continue;
        }

        // 导入内存
        xchg_mem_info_t mem_info = { ... };
        imported_tseg = import_mem(queue->dev_ctx->urma_ctx, &mem_info);

        // 发送 IMPORT_DONE 确认
        umq_ub_import_mem_done(queue, import_mempool_info[i].mempool_id);
        //    └─ umq_ub_write_imm(IMM_TYPE_MEM_IMPORT_DONE)
    }
}
